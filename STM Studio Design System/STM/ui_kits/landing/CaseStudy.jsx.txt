const { NavBar, Eyebrow, Frame, CreditRow, Rule, Button, Footer } = window.STMStudioDesignSystem_5c3910;

function CaseStudy({ work, onBack, onNext, lang = 'en', onLang }) {
  const C = window.STM_COPY;
  const w = work || C.work[0];
  const idx = C.work.findIndex((x) => x.frame === w.frame);
  const next = C.work[(idx + 1) % C.work.length];

  return (
    <div style={{ width: 1440, margin: '0 auto', background: 'var(--stm-ink)', color: 'var(--stm-paper)', fontFamily: 'var(--stm-font-body)' }}>
      <NavBar lang={lang} onLang={onLang} links={[{ label: 'Work' }, { label: 'Services' }, { label: 'The Unit' }, { label: 'Studio' }]} />
      <div style={{ padding: '56px 100px 0' }}>
        <button onClick={onBack} style={{
          background: 'none', border: 0, padding: 0, cursor: 'pointer', display: 'flex', alignItems: 'center', gap: 8,
          fontFamily: 'var(--stm-font-mono)', fontSize: 11, letterSpacing: '0.16em', textTransform: 'uppercase', color: 'var(--stm-grey-6)'
        }}>← All work</button>
        <div style={{ display: 'flex', gap: 90, marginTop: 30, alignItems: 'flex-start' }}>
          <div style={{ flexGrow: 1 }}>
            <Eyebrow>{w.credits[0].value}</Eyebrow>
            <h1 style={{ fontFamily: 'var(--stm-font-display)', fontSize: 64, lineHeight: 0.98, fontWeight: 700, letterSpacing: '-0.03em', margin: '16px 0 0 0' }}>{w.frame.split('· ')[1] || w.credits[0].value}</h1>
          </div>
          <div style={{ width: 320, flexShrink: 0, display: 'flex', flexDirection: 'column', gap: 12, paddingTop: 10 }}>
            {w.credits.map((c) => <CreditRow key={c.label} label={c.label} value={c.value} accent={c.accent} />)}
          </div>
        </div>
        <Frame style={{ marginTop: 50 }} height={640} label="Hero frame" note="still to supply" badge="CASE STUDY" />
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(2, minmax(0, 1fr))', gap: 24, marginTop: 24 }}>
          <Frame height={360} label="Gallery 01" note="still to supply" />
          <Frame height={360} label="Gallery 02" note="still to supply" />
        </div>
      </div>
      <div style={{ marginTop: 110, borderTop: '1px solid var(--stm-line-14)', padding: '40px 100px', display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
        <span style={{ fontFamily: 'var(--stm-font-mono)', fontSize: 11, letterSpacing: '0.16em', color: 'var(--stm-grey-7)', textTransform: 'uppercase' }}>Next project</span>
        <button onClick={() => onNext && onNext(next)} style={{
          background: 'none', border: 0, padding: 0, cursor: 'pointer',
          fontFamily: 'var(--stm-font-display)', fontSize: 24, fontWeight: 600, letterSpacing: '-0.015em', color: 'var(--stm-paper)'
        }}>{next.credits[0].value} →</button>
      </div>
      <Footer />
    </div>
  );
}
Object.assign(window, { CaseStudy });
