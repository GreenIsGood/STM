export function ServiceRow({ number, title, children, mobile = false, arabic = false, style }) {

      <div style={{ padding: '22px 0', ...style }}>
        <div style={{ display: 'flex', gap: 14, alignItems: 'baseline' }}>

          <h3 style={{ fontFamily: 'var(--stm-font-display)', fontSize: 21, fontWeight: 600, letterSpacing: '-0.015em', color: 'var(--stm-paper)', margin: 0 }}>{title}</h3>

      </div>
    );
  }
  return (
    <div style={{ display: 'flex', gap: 48, padding: '34px 0', alignItems: 'flex-start', ...style }}>
      <span style={{ fontFamily: 'var(--stm-font-mono)', fontSize: 13, color: 'var(--stm-red)', width: 46, flexShrink: 0, paddingTop: 7 }}>{number}</span>
      <h3 style={{
        fontFamily: arabic ? 'var(--stm-font-arabic)' : 'var(--stm-font-display)',

        color: 'var(--stm-paper)', margin: 0, width: 420, flexShrink: 0
      }}>{title}</h3>

      }}>{children}</p>
    </div>
  );
}
